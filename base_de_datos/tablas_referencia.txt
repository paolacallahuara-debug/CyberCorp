-- ============================================================
-- 2. FUNCIÓN PARA ACTUALIZAR updated_at AUTOMÁTICAMENTE
-- ============================================================

CREATE OR REPLACE FUNCTION public.actualizar_updated_at()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$;


-- ============================================================
-- 3. TABLA: ADMINISTRADORES
-- El usuario/contraseña se maneja mediante Supabase Auth.
-- Esta tabla solamente contiene el perfil y rol.
-- ============================================================

CREATE TABLE public.administradores (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    auth_user_id UUID NOT NULL UNIQUE
        REFERENCES auth.users(id)
        ON DELETE CASCADE,

    nombre VARCHAR(100) NOT NULL,
    email VARCHAR(150) NOT NULL UNIQUE,

    rol VARCHAR(30) NOT NULL DEFAULT 'admin'
        CHECK (rol IN ('admin', 'super_admin')),

    activo BOOLEAN NOT NULL DEFAULT TRUE,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_administradores_auth_user
ON public.administradores(auth_user_id);


-- ============================================================
-- 4. TABLA: CATEGORIAS
-- ============================================================

CREATE TABLE public.categorias (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    nombre VARCHAR(100) NOT NULL UNIQUE,
    slug VARCHAR(120) NOT NULL UNIQUE,

    descripcion TEXT,
    imagen_url TEXT,

    activo BOOLEAN NOT NULL DEFAULT TRUE,

    orden INTEGER NOT NULL DEFAULT 0
        CHECK (orden >= 0),

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


-- ============================================================
-- 5. TABLA: MARCAS
-- ============================================================

CREATE TABLE public.marcas (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    nombre VARCHAR(100) NOT NULL UNIQUE,
    slug VARCHAR(120) NOT NULL UNIQUE,

    logo_url TEXT,

    activo BOOLEAN NOT NULL DEFAULT TRUE,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()

-- ============================================================
-- 6. TABLA: PRODUCTOS
-- ============================================================

CREATE TABLE public.productos (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    codigo VARCHAR(100) NOT NULL UNIQUE,

    nombre VARCHAR(200) NOT NULL,

    slug VARCHAR(250) NOT NULL UNIQUE,

    descripcion_corta VARCHAR(300),
    descripcion TEXT,

    categoria_id BIGINT
        REFERENCES public.categorias(id)
        ON DELETE SET NULL,

    marca_id BIGINT
        REFERENCES public.marcas(id)
        ON DELETE SET NULL,

    precio NUMERIC(12,2) NOT NULL DEFAULT 0
        CHECK (precio >= 0),

    precio_anterior NUMERIC(12,2)
        CHECK (
            precio_anterior IS NULL
            OR precio_anterior >= 0
        ),

    stock INTEGER NOT NULL DEFAULT 0
        CHECK (stock >= 0),

    destacado BOOLEAN NOT NULL DEFAULT FALSE,

    nuevo BOOLEAN NOT NULL DEFAULT FALSE,

    activo BOOLEAN NOT NULL DEFAULT TRUE,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


CREATE INDEX idx_productos_categoria
ON public.productos(categoria_id);

CREATE INDEX idx_productos_marca
ON public.productos(marca_id);

CREATE INDEX idx_productos_activo
ON public.productos(activo);

CREATE INDEX idx_productos_destacado
ON public.productos(destacado);

CREATE INDEX idx_productos_nombre
ON public.productos(nombre);

CREATE TABLE public.kits (

    codigo VARCHAR(100) NOT NULL UNIQUE,

    nombre VARCHAR(200) NOT NULL,

    slug VARCHAR(250) NOT NULL UNIQUE,

    descripcion_corta VARCHAR(300),

    descripcion TEXT,

    precio NUMERIC(12,2) NOT NULL DEFAULT 0
        CHECK (precio >= 0),

    precio_anterior NUMERIC(12,2)
        CHECK (
            precio_anterior IS NULL
            OR precio_anterior >= 0

-- 14. FUNCIÓN PARA SABER SI EL USUARIO ES ADMINISTRADOR
-- ============================================================

CREATE OR REPLACE FUNCTION public.es_administrador()
RETURNS BOOLEAN
LANGUAGE sql
SECURITY DEFINER
SET search_path = public
STABLE
AS $$
    SELECT EXISTS (
        SELECT 1
        FROM public.administradores
        WHERE auth_user_id = auth.uid()
        AND activo = TRUE
    );
$$;


-- Evita acceso innecesario directo a la función.

REVOKE ALL ON FUNCTION public.es_administrador() FROM PUBLIC;

GRANT EXECUTE ON FUNCTION public.es_administrador()
TO anon, authenticated;

-- 1) CAMPOS NUEVOS EN TABLAS EXISTENTES

  ADD COLUMN IF NOT EXISTS fecha_fin TIMESTAMPTZ;

  ADD COLUMN IF NOT EXISTS ubicacion_url TEXT;

-- Evita rangos de fechas invertidos. Los bloques comprueban si la
-- restricción ya existe antes de crearla.
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint
    WHERE conname = 'cursos_fechas_validas'
      AND conrelid = 'public.cursos'::regclass
  ) THEN
    ALTER TABLE public.cursos
      ADD CONSTRAINT cursos_fechas_validas
      CHECK (fecha_fin IS NULL OR fecha_inicio IS NULL OR fecha_fin >= fecha_inicio);
  END IF;
END $$;

DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint
    WHERE conname = 'kits_fechas_validas'
      AND conrelid = 'public.kits'::regclass
  ) THEN
    ALTER TABLE public.kits
      ADD CONSTRAINT kits_fechas_validas
      CHECK (fecha_fin IS NULL OR fecha_inicio IS NULL OR fecha_fin >= fecha_inicio);
  END IF;
END $$;

-- 2) CONVENIOS
CREATE TABLE IF NOT EXISTS public.convenios (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  titulo TEXT NOT NULL,
  descripcion TEXT,
  imagen_url TEXT,
  enlace_url TEXT,
  activo BOOLEAN NOT NULL DEFAULT TRUE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_convenios_activo ON public.convenios(activo);

-- 3) TESTIMONIOS
CREATE TABLE IF NOT EXISTS public.testimonios (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  titulo TEXT NOT NULL,
  nombre_cliente TEXT,
  descripcion TEXT,
  tipo_video TEXT NOT NULL DEFAULT 'url'
    CHECK (tipo_video IN ('archivo', 'url')),
  video_url TEXT,
  video_externo_url TEXT,
  activo BOOLEAN NOT NULL DEFAULT TRUE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  CHECK (
    (tipo_video = 'archivo' AND video_url IS NOT NULL)
    OR
    (tipo_video = 'url' AND video_externo_url IS NOT NULL)
  )
);

CREATE INDEX IF NOT EXISTS idx_testimonios_activo ON public.testimonios(activo);

  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_tiktok_publicaciones_activo
  ON public.tiktok_publicaciones(activo);
